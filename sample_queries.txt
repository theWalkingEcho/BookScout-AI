================================================================================
  BOOKSCOUT AI - SAMPLE API QUERIES
  POST /chat   |   Content-Type: application/json
================================================================================

Schema:
  {
    "query": "<natural-language question>",
    "history": [
      { "role": "user",      "content": "<previous user turn>" },
      { "role": "assistant", "content": "<previous assistant turn>" }
    ]
  }

  - "query"   : required - the current user question
  - "history" : optional - list of prior conversation turns (oldest first)
  - "role"    : either "user" or "assistant"

================================================================================
  1. SIMPLE QUERIES  (no history)
================================================================================

-- 1a. Browse by genre --
{
  "query": "Show me all available Science Fiction books",
  "history": []
}

{

}

================================================================================
  2. MULTI-TURN QUERIES  (with conversation history)
================================================================================

-- 2a. Follow-up: narrow down results --
{
  "query": "Which of those are under LKR 1500?",
  "history": [
    {
      "role": "user",
      "content": "Show me all Mystery books"
    },
    {
      "role": "assistant",
      "content": "Here are the Mystery books in our inventory: 'The Girl with the Dragon Tattoo', 'Gone Girl', 'Big Little Lies', and 'The Silent Patient'. Prices range from LKR 900 to LKR 2200."
    }
  ]
}

-- 2b. Follow-up: ask for more details --
{
  "query": "Tell me more about The Silent Patient",
  "history": [
    {
      "role": "user",
      "content": "List some popular psychological thrillers"
    },
    {
      "role": "assistant",
      "content": "Popular psychological thrillers in stock include 'The Silent Patient', 'Behind Closed Doors', and 'The Woman in the Window'."
    }
  ]
}

-- 2c. Follow-up: compare two items --
{
  "query": "Which one is cheaper and which has better stock?",
  "history": [
    {
      "role": "user",
      "content": "Do you have Dune by Frank Herbert?"
    },
    {
      "role": "assistant",
      "content": "Yes! We have 'Dune' (LKR 1800, 12 copies) and 'Dune Messiah' (LKR 1600, 5 copies) by Frank Herbert."
    }
  ]
}

-- 2d. Multi-turn: three turns deep --
{
  "query": "Can you also filter to show only paperback editions?",
  "history": [
    {
      "role": "user",
      "content": "What self-help books do you stock?"
    },
    {
      "role": "assistant",
      "content": "We carry 'Atomic Habits', 'The 7 Habits of Highly Effective People', 'Think and Grow Rich', and 'The Power of Now'."
    },
    {
      "role": "user",
      "content": "Show me only the ones under LKR 2000"
    },
    {
      "role": "assistant",
      "content": "'Atomic Habits' (LKR 1750) and 'Think and Grow Rich' (LKR 1200) are both under LKR 2000."
    }
  ]
}
